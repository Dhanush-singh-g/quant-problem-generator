$exclude = @(
    ".env",
    ".env.local",
    ".env.production",
    ".env.development",
    "node_modules",
    ".next",
    "__pycache__",
    ".git",
    ".venv",
    "venv",
    "dist",
    "build"
)

$files = Get-ChildItem -Recurse -File | Where-Object {
    $path = $_.FullName

    -not (
        $exclude | Where-Object {
            $path -match "(^|\\)$([regex]::Escape($_))(\\|$)"
        }
    ) -and

    $_.Name -notmatch "^\.env" -and
    $_.Extension -notin @(".pyc", ".log")
}

$files | Compress-Archive -DestinationPath "../QuantForge-review.zip" -Force
